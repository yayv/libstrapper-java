# libstrapper 内置扩展类型表
#
# 每行一条:  <基础类型>.<扩展名>  :  <正则>
# 以 # 开头的行是注释, 空行忽略. 名字与正则之间用第一个 ':' 或 '=' 分隔.
#
# 基础类型固定为这 9 个, 它们决定取值最终被解析成什么:
#   int float number  -> 数字      bool -> 布尔
#   string text       -> 字符串    date time datetime -> 归一化后的日期/时间串
#
# 一个值要先过基础类型的正则, 再过这里写的正则, 两关都过才算合格.
# 同名再写一次就是覆盖; 想改内置规则, 在自己的类型表里重写同名一行即可.
#
# 关于 string 和 text 的分工:
#   string 是纯字符串, 不允许出现制表符、换行符这类转义字符;
#   text 允许. 所以凡是单行的短串一律挂在 string 下面, 只有确实可能带换行的
#   才挂到 text 下面 —— 目前只有 inlineImage 一条.
#   这样调用方可以用 disableType("text") 一次性关掉全部可带换行的类型,
#   需要收紧攻击面时很有用; 代价是 inlineImage 也跟着不能用了.
#
# 关于正则方言:
#   这张表是给各语言实现共用的, 所以只用各家正则库都认的写法 ——
#   ^ $ 锚定, [] 字符组, () 与 (?:) 分组, | 选择, * + ? {n,m} 量词, \d \s \x21 转义.
#   刻意不用: 内联修饰组 (?i:...)、命名组、反向引用、断言 —— 这些在 JS / Go RE2 上支持不一.
#   移植到别的语言时如果某条不合用, 照着自己语言的正则库改这一行就行, 不影响其他类型.

# ---- 整数类型的扩展 ----
int.year           : ^\d{4}$
int.month          : ^(0?[1-9]|1[0-2])$
int.day            : ^(0?[1-9]|[12]\d|3[01])$
int.age            : ^\d{1,3}$

# ---- 小数类型的扩展，不区分定点和浮点 -----
float.currency     : ^[+-]?\d+(\.\d{1,2})?$

# ---- 日期类型的扩展 -----
#date.31days     : ^[+-]?\d+(\.\d{1,2})?$

# ---- 时间类型的扩展 -----
#time.currency     : ^[+-]?\d+(\.\d{1,2})?$

# ---- 日期时间类型的扩展 -----
#datetime.currency     : ^[+-]?\d+(\.\d{1,2})?$

# ---- 带格式或符号的数字 ----
string.phone       : ^\+?[0-9][0-9\- ]{4,18}[0-9]$
string.mobile      : ^(\+?86[- ]?)?1[3-9]\d{9}$

# ---- 有特殊含义的常见字母组合 ----
string.weekday     : ^([Mm]on(day)?|[Tt]ue(sday)?|[Ww]ed(nesday)?|[Tt]hu(rsday)?|[Ff]ri(day)?|[Ss]at(urday)?|[Ss]un(day)?)$

# ---- 对字符有一定约束 ----
string.lower       : ^[a-z]+$
string.upper       : ^[A-Z]+$
string.letter      : ^[A-Za-z]+$

# ---- 编码格式 ----
string.idcard      : ^(\d{15}|\d{17}[0-9Xx])$
string.plateNumber : ^[京津沪渝冀豫云辽黑湘皖鲁新苏浙赣鄂桂甘晋蒙陕吉闽贵粤青藏川宁琼][A-HJ-NP-Z][A-HJ-NP-Z0-9]{4,5}[A-HJ-NP-Z0-9挂学警港澳使领]$
string.verify      : ^[0-9A-Za-z]{4,8}$
string.retCode     : ^(ok|fail|error|deny)$
string.MD5         : ^[0-9a-fA-F]{32}$
string.base64      : ^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$
string.email       : ^[A-Za-z0-9!#$%&'*+/=?^_`{|}~.-]+@[A-Za-z0-9]([A-Za-z0-9-]*[A-Za-z0-9])?(\.[A-Za-z0-9]([A-Za-z0-9-]*[A-Za-z0-9])?)+$

# ---- 唯一挂在 text 下的类型: data URI 可能很长甚至带换行 ----
text.inlineImage   : ^data:image/(png|jpe?g|gif|webp|bmp|svg\+xml);base64,[A-Za-z0-9+/=\s]+$

# ---- 有格式要求和一定顺序要求的字母数字符号组合 ----
string.username    : ^[A-Za-z][A-Za-z0-9_.-]{2,31}$
string.password    : ^[\x21-\x7E]{6,64}$
